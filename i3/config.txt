mode "workspaces" {
    bindsym 1  workspace $ws1
    bindsym 2  workspace $ws2
    bindsym 3  workspace $ws3
    bindsym 4  workspace $ws4

    bindsym h  workspace next
    bindsym l  workspace prev
    bindsym j  workspace back_and_forth

    
    bindsym Shift+1  move container to workspace $ws1
    bindsym Shift+2  move container to workspace $ws2
    bindsym Shift+3  move container to workspace $ws3
    bindsym Shift+4  move container to workspace $ws4
    
    bindsym Shift+h  move container to workspace next
    bindsym Shift+l  move container to workspace prev
    bindsym Shift+j  move container to workspace back_and_forth


    bindsym $alt+q  mode "default"
}

bindsym $alt+w  mode "workspaces"

# ---Resize and moving---
mode "resizer" {
    bindsym h  resize shrink width  10 px
    bindsym j  resize grow   height 10 px
    bindsym k  resize shrink height 10 px
    bindsym l  resize grow   width  10 px

    bindsym Shift+h  resize shrink width  1 px
    bindsym Shift+j  resize grow   height 1 px
    bindsym Shift+k  resize shrink height 1 px
    bindsym Shift+l  resize grow   width  1 px


    bindsym $alt+h  move left  10 px 
    bindsym $alt+k  move up    10 px
    bindsym $alt+j  move down  10 px
    bindsym $alt+l  move right 10 px

    bindsym $alt+Shift+h  move left  1 px 
    bindsym $alt+Shift+k  move up    1 px
    bindsym $alt+Shift+j  move down  1 px
    bindsym $alt+Shift+l  move right 1 px


    bindsym $alt+q  mode "default"
}

bindsym $alt+r  mode "resizer"
